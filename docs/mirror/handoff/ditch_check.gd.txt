extends SceneTree
const GeneratorScript := preload("res://src/worldgen/runtime/terrain_generator.gd")
const WorldSettingsScript := preload("res://src/worldgen/settings/world_settings.gd")
const TileHeightScript := preload("res://src/worldgen/tile/tile_height.gd")
const TableSetScript := preload("res://src/worldgen/tables/table_set.gd")
const WaterLive := preload("res://src/water/water_live.gd")
var g
var data
var spots: Array = []   # [Vector3]

func make_live() -> Object:
	var live = WaterLive.new()
	live.setup(Vector2i(3, 4), data, "/tmp/ditch.water", 3.0, g, true)
	return live

func run_until(live, sim_s: float) -> void:
	while live.sim.seconds < sim_s:
		live.wake(100000.0)
		live.tick(10.0)
		if live._since == 0:
			live.take_levels()

func ditch_report(live, label: String) -> void:
	var sim = live.sim
	var side: int = data.samples
	var wet := 0
	var deep := 0
	for s in spots:
		var i := int(round((s.x - data.x0) / 8.0))
		var j := int(round((s.z - data.z0) / 8.0))
		var d: float = sim.depth_at(j * side + i)
		if d > 0.1: wet += 1
		if d > 0.5: deep += 1
	print("%s: %d of %d ditch cells wet (>10 cm), %d deeper than 50 cm, sim %.0f s, total %.0f m3" % [label, wet, spots.size(), deep, sim.seconds, sim.total_volume()])

func _init() -> void:
	var errors: Array = []
	var settings = WorldSettingsScript.load_from_file("res://worlds/base_world/world.json", errors)
	var rules = TileHeightScript.load_rules("res://worlds/base_world/plan/tile_rules.json", errors)
	var tables = TableSetScript.new()
	errors.append_array(tables.load_directory())
	g = GeneratorScript.new()
	g.setup(settings, rules, tables, "res://worlds/base_world/tiles")
	data = g.get_tile(Vector2i(3, 4))
	data.sim_levels = PackedFloat32Array()
	data.sim_levels.resize(data.samples * data.samples); data.sim_levels.fill(NAN)
	var side: int = data.samples
	# the river cell with the biggest catchment within 500 m of the spawn (the starter river)
	var spawn := Vector2(53233.0, 68416.0)
	var best := -1.0
	var bi := 0
	var bj := 0
	for dj in range(-62, 63):
		for di in range(-62, 63):
			var i := int((spawn.x - data.x0) / 8.0) + di
			var j := int((spawn.y - data.z0) / 8.0) + dj
			var k := j * side + i
			if data.water_class[k] == 3 and data.heights[k] > 3.0 and data.flow[k] > best:
				best = data.flow[k]; bi = i; bj = j
	print("river cell at ", Vector2(data.x0 + bi * 8.0, data.z0 + bj * 8.0), " flow ", best, " height ", data.heights[bj * side + bi])
	# a direction away from the river: the neighbour direction whose ground rises most within 4 cells (but not water)
	var dirs := [Vector2i(1, 0), Vector2i(-1, 0), Vector2i(0, 1), Vector2i(0, -1)]
	var pick := Vector2i(1, 0)
	var pick_score := 1e9
	for d in dirs:
		var ok := true
		var rise := 0.0
		for n in range(2, 8):
			var k: int = (bj + d.y * n) * side + bi + d.x * n
			if data.water_class[k] != 0: ok = false
			rise = data.heights[k] - data.heights[bj * side + bi]
		if ok and absf(rise) < pick_score:
			pick_score = absf(rise); pick = d
	print("ditch direction ", pick, " rise over 6 cells ", pick_score)
	# 6 digs of radius 3 along the line from 2 cells out, every 4 m
	var digs: Array = []
	for n in range(0, 7):
		var x: float = data.x0 + float(bi) * 8.0 + float(pick.x) * (16.0 + 4.0 * n)
		var z: float = data.z0 + float(bj) * 8.0 + float(pick.y) * (16.0 + 4.0 * n)
		digs.append(Vector3(x, g.surface_height(x, z), z))
	var seen := {}
	for dg in digs:
		for dx in range(-1, 2):
			for dz in range(-1, 2):
				var i := int(round((dg.x - data.x0) / 8.0)) + dx
				var j := int(round((dg.z - data.z0) / 8.0)) + dz
				seen[j * side + i] = true
	for k in seen:
		spots.append(Vector3(data.x0 + (k % side) * 8.0, 0.0, data.z0 + (k / side) * 8.0))
	# Run A: dig live after the rivers run
	var a = make_live()
	a.begin()
	run_until(a, 2400.0)
	ditch_report(a, "before the dig")
	for dg in digs: a.dig(dg, 3.0)
	run_until(a, 2400.0 + 300.0)
	ditch_report(a, "300 s after the dig")
	run_until(a, 2400.0 + 1800.0)
	ditch_report(a, "1800 s after the dig")
	# Run B: the same digs replayed before the first tick (a restart)
	var b = make_live()
	b.begin()
	for dg in digs: b.dig(dg, 3.0)
	run_until(b, 2400.0 + 1800.0)
	ditch_report(b, "restart replay, same simulated time")
	quit()
